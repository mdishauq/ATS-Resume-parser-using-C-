@app.route('/generate', methods=['POST'])
def generate_resume():
    try:
        resume_data = request.json
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        json_filename = f'resume_{timestamp}.json'
        pdf_filename = f'resume_{timestamp}.pdf'
        
        json_path = os.path.join(UPLOAD_FOLDER, json_filename)
        pdf_path = os.path.join(OUTPUT_FOLDER, pdf_filename)
        
        with open(json_path, 'w') as f:
            json.dump(resume_data, f, indent=2)
        
        result = subprocess.run(
            [CPP_ENGINE_PATH, json_path, pdf_path],
            capture_output=True,
            text=True,
            timeout=30
        )
        
        if result.returncode != 0:
            return jsonify({'success': False, 'error': f'Engine error: {result.stderr}'}), 500
        
        if not os.path.exists(pdf_path):
            return jsonify({'success': False, 'error': 'PDF was not generated'}), 500
        
        return jsonify({'success': True, 'filename': pdf_filename})
    
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)}), 500

@app.route('/preview/<filename>')
def preview_file(filename):
    """Preview PDF in browser"""
    try:
        file_path = os.path.join(OUTPUT_FOLDER, filename)
        if not os.path.exists(file_path):
            return jsonify({'error': 'File not found'}), 404
        return send_file(file_path, mimetype='application/pdf')
    except Exception as e:
        return jsonify({'error': str(e)}), 500

@app.route('/download/<filename>')
def download_file(filename):
    """Download generated PDF"""
    try:
        file_path = os.path.join(OUTPUT_FOLDER, filename)
        if not os.path.exists(file_path):
            return jsonify({'error': 'File not found'}), 404
        return send_file(file_path, as_attachment=True, download_name=f'ATS_Resume_{filename}')
    except Exception as e:
        return jsonify({'error': str(e)}), 500

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
